Embed on Your Website

Copy and paste any of these snippets directly into your frontend or backend: 

// 1. Fetch live telemetry using X-API-Key header
async function getTemperature() {
  const res = await fetch('http://localhost:8000/v1/telemetry/live', {
    headers: { 'X-API-Key': 'chillpath_live_secret_key' }
  });
  const json = await res.json();
  console.log('Temperature:', json.data.temperature_c, '°C');
  return json.data;
}

// 2. Or simply use the URL query parameter:
// http://localhost:8000/v1/telemetry/live?api_key=chillpath_live_secret_key
# Using X-API-Key Header:
curl -X GET "http://localhost:8000/v1/telemetry/live" \
     -H "X-API-Key: chillpath_live_secret_key"

# Or via query parameter:
curl "http://localhost:8000/v1/telemetry/live?api_key=chillpath_live_secret_key"
import requests

url = "http://localhost:8000/v1/telemetry/live"
headers = {"X-API-Key": "chillpath_live_secret_key"}

data = requests.get(url, headers=headers).json()
print(f"Current Temp: {data['data']['temperature_c']} °C")
// Trigger an SMS to +919557066830 from any webpage
fetch('http://localhost:8000/v1/telemetry/sms/send', {
  method: 'POST',
  headers: {
    'Content-Type': 'application/json',
    'X-API-Key': 'chillpath_live_secret_key'
  },
  body: JSON.stringify({
    phone_number: '+919557066830',
    message: 'ChillPath Alert: Temperature threshold exceeded!'
  })
});
